    83→// ==================== Helpers ====================
    84→/** Current IST month as YYYY-MM, offset by n months */
    85→function ymIST(offsetMonths = 0): string {
    86→  const [y, m] = ymdIST().split("-").map(Number)
    87→  let year = y, month = m + offsetMonths
    88→  while (month < 1) { month += 12; year -= 1 }
    89→  while (month > 12) { month -= 12; year += 1 }
    90→  return `${year}-${String(month).padStart(2, "0")}`
    91→}
    92→
    93→function isTaskOverdue(t: Task): boolean {
    94→  if (!t.dueDate || t.status === "COMPLETED" || t.status === "CANCELLED") return false
    95→  return fmtDateIST(t.dueDate, "yyyy-MM-dd") < ymdIST()
    96→}

   279→// ==================== EMPLOYEE FORM (create / edit) ====================
   280→function EmployeeForm({ employee, onClose }: { employee?: Employee; onClose: () => void }) {
   281→  const qc = useQueryClient()
   282→  const [saving, setSaving] = useState(false)
   283→  const [form, setForm] = useState({
   284→    name: employee?.name ?? "",
   285→    phone: employee?.phone ?? "",
   286→    designation: employee?.designation ?? "",
   287→    joiningDate: employee?.joiningDate ? fmtDateIST(employee.joiningDate, "yyyy-MM-dd") : ymdIST(),
   288→    salary: employee?.salary ?? 0,
   289→    address: employee?.address ?? "",
   290→    notes: employee?.notes ?? "",
   291→  })
   292→
   293→  async function save() {
   294→    if (!form.name.trim()) return toast({ title: "Name is required", variant: "destructive" })
   295→    setSaving(true)
   296→    try {
   297→      const payload = {
   298→        name: form.name.trim(),
   299→        phone: form.phone.trim(),
   300→        designation: form.designation.trim(),
   301→        joiningDate: form.joiningDate || undefined,
   302→        salary: form.salary,
   303→        address: form.address.trim(),
   304→        notes: form.notes.trim(),
   305→      }
   306→      if (employee) await api.put(`staff/${employee.id}`, payload)
   307→      else await api.post("staff", payload)
   308→      toast({
   309→        title: employee ? "Employee updated" : "Employee created",
   310→        description: `${form.name.trim()}${form.designation.trim() ? ` · ${form.designation.trim()}` : ""}`,
   311→      })
   312→      qc.invalidateQueries({ queryKey: ["staff"] })
   313→      onClose()
   314→    } catch (e: any) {
   315→      toast({ title: "Failed to save employee", description: e.message, variant: "destructive" })
   316→    } finally {
   317→      setSaving(false)
   318→    }
   319→  }
   320→
   321→  return (
   322→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   323→      <DialogContent className="max-w-lg">
   324→        <DialogHeader>
   325→          <DialogTitle>{employee ? `Edit ${employee.name}` : "New Employee"}</DialogTitle>
   326→        </DialogHeader>
   327→        <div className="grid gap-3 sm:grid-cols-2">
   328→          <div className="sm:col-span-2">
   329→            <Field label="Full name" required>
   330→              <TextInput value={form.name} onChange={(v) => setForm({ ...form, name: v })} placeholder="Employee name" autoFocus />
   331→            </Field>
   332→          </div>
   333→          <Field label="Phone">
   334→            <TextInput value={form.phone} onChange={(v) => setForm({ ...form, phone: v })} placeholder="98XXXXXXXX" />
   335→          </Field>
   336→          <Field label="Designation" hint="e.g. Sales Person, Tailor, Helper">
   337→            <TextInput value={form.designation} onChange={(v) => setForm({ ...form, designation: v })} placeholder="Designation" />
   338→          </Field>
   339→          <Field label="Joining date" hint={employee ? "Server keeps the original joining date for now" : undefined}>
   340→            <TextInput type="date" value={form.joiningDate} onChange={(v) => setForm({ ...form, joiningDate: v })} />
   341→          </Field>
   342→          <Field label="Monthly salary (₹)">
   343→            <NumberInput value={form.salary} onChange={(v) => setForm({ ...form, salary: v })} min={0} />
   344→          </Field>
   345→          <div className="sm:col-span-2">
   346→            <Field label="Address">
   347→              <TextInput value={form.address} onChange={(v) => setForm({ ...form, address: v })} placeholder="Street, city…" />
   348→            </Field>
   349→          </div>
   350→          <div className="sm:col-span-2">
   351→            <Field label="Notes">
   352→              <TextArea value={form.notes} onChange={(v) => setForm({ ...form, notes: v })} rows={2} placeholder="Remarks, skills, references…" />
   353→            </Field>
   354→          </div>
   355→        </div>
   356→        <DialogFooter>
   357→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   358→          <Button onClick={save} disabled={saving}>
   359→            {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}Save
   360→          </Button>
   361→        </DialogFooter>
   362→      </DialogContent>
   363→    </Dialog>
   364→  )
   365→}

   367→// ==================== EMPLOYEE DETAIL SHEET ====================
   368→function EmployeeDetailSheet({
   369→  employeeId, employee, notFound, onClose, onEdit, onMarkAttendance,
   370→}: {
   371→  employeeId: string
   372→  employee: Employee | null
   373→  notFound?: boolean
   374→  onClose: () => void
   375→  onEdit: (e: Employee) => void
   376→  onMarkAttendance: () => void
   377→}) {
   378→  const qc = useQueryClient()
   379→  const [payOpen, setPayOpen] = useState(false)
   380→  const [deactivateOpen, setDeactivateOpen] = useState(false)
   381→  const [togglingStatus, setTogglingStatus] = useState(false)
   382→
   383→  // Salary / advance history (documented endpoint: GET staff/payments?employeeId=)
   384→  const { data: payData, isLoading: payLoading } = useQuery({
   385→    queryKey: ["staff", "payments", employeeId],
   386→    queryFn: () => api.get(`staff/payments${qs({ employeeId })}`),
   387→  })
   388→
   389→  // Recent attendance = current + previous month via GET staff/attendance?month=
   390→  const curMonth = useMemo(() => ymIST(0), [])
   391→  const prevMonth = useMemo(() => ymIST(-1), [])
   392→  const { data: attCur } = useQuery({
   393→    queryKey: ["staff", "attendance", "month", curMonth],
   394→    queryFn: () => api.get(`staff/attendance${qs({ month: curMonth })}`),
   395→  })
   396→  const { data: attPrev } = useQuery({
   397→    queryKey: ["staff", "attendance", "month", prevMonth],
   398→    queryFn: () => api.get(`staff/attendance${qs({ month: prevMonth })}`),
   399→  })
   400→
   401→  const attendanceRecords: AttendanceRecord[] = useMemo(() => {
   402→    const all: AttendanceRecord[] = [...(attCur?.records ?? []), ...(attPrev?.records ?? [])]
   403→    return all
   404→      .filter((r) => r.employeeId === employeeId)
   405→      .sort((a, b) => b.date.localeCompare(a.date))
   406→      .slice(0, 60)
   407→  }, [attCur, attPrev, employeeId])
   408→
   409→  const payments: SalaryPayment[] = payData?.payments ?? []
   410→  const presentCount = attendanceRecords.filter((r) => r.status === "PRESENT").length
   411→  const totalPaid = payments.reduce((s, p) => s + Math.max(0, p.amount), 0)
   412→  const advanceTotal = payments.filter((p) => p.type === "ADVANCE").reduce((s, p) => s + p.amount, 0)
   413→
   414→  async function updateStatus(id: string, status: "ACTIVE" | "INACTIVE") {
   415→    setTogglingStatus(true)
   416→    try {
   417→      await api.put(`staff/${id}`, { status })
   418→      toast({
   419→        title: status === "INACTIVE" ? "Employee deactivated" : "Employee reactivated",
   420→        description: employee?.name ?? "",
   421→      })
   422→      qc.invalidateQueries({ queryKey: ["staff"] })
   423→      if (status === "INACTIVE") onClose()
   424→    } catch (e: any) {
   425→      toast({ title: "Could not update employee", description: e.message, variant: "destructive" })
   426→    } finally {
   427→      setTogglingStatus(false)
   428→      setDeactivateOpen(false)
   429→    }
   430→  }
   431→
   432→  if (!employee) {
   433→    return (
   434→      <Sheet open onOpenChange={(v) => !v && onClose()}>
   435→        <SheetContent side="right" className="w-full overflow-y-auto p-0 sm:max-w-2xl thin-scrollbar">
   436→          <SheetHeader className="border-b bg-muted/40 px-5 py-4">
   437→            <SheetTitle>Employee</SheetTitle>
   438→          </SheetHeader>
   439→          <div className="flex items-center justify-center py-24">
   440→            {notFound ? (
   441→              <EmptyState
   442→                title="Employee not found"
   443→                description="This employee may have been removed. Close and refresh the list."
   444→                action={<Button size="sm" variant="outline" onClick={onClose}>Close</Button>}
   445→              />
   446→            ) : (
   447→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   448→            )}
   449→          </div>
   450→        </SheetContent>
   451→      </Sheet>
   452→    )
   453→  }
   454→
   455→  return (
   456→    <Sheet open onOpenChange={(v) => !v && onClose()}>
   457→      <SheetContent side="right" className="w-full overflow-y-auto p-0 sm:max-w-2xl thin-scrollbar">
   458→        <SheetHeader className="border-b bg-muted/40 px-5 py-4">
   459→          <SheetTitle className="flex items-center gap-3">
   460→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-primary text-lg font-bold text-primary-foreground">

   625→// ==================== PAY SALARY / ADVANCE DIALOG ====================
   626→function PaySalaryDialog({ employee, onClose }: { employee: Employee; onClose: () => void }) {
   627→  const qc = useQueryClient()
   628→  const [saving, setSaving] = useState(false)
   629→  const [form, setForm] = useState({
   630→    type: "SALARY",
   631→    amount: employee.salary || 0,
   632→    method: "CASH",
   633→    month: ymIST(0),
   634→    date: ymdIST(),
   635→    notes: "",
   636→  })
   637→
   638→  function changeType(type: string) {
   639→    setForm((f) => ({ ...f, type, amount: type === "SALARY" ? (employee.salary || 0) : f.amount }))
   640→  }
   641→
   642→  async function save() {
   643→    if (!form.amount || form.amount <= 0) {
   644→      return toast({ title: "Enter a valid amount", variant: "destructive" })
   645→    }
   646→    setSaving(true)
   647→    try {
   648→      await api.post("staff/payments", { employeeId: employee.id, ...form })
   649→      toast({
   650→        title: `${SALARY_PAYMENT_TYPE_LABELS[form.type] ?? form.type} recorded`,
   651→        description: `${employee.name} — ${fmtMoney(form.amount)} via ${PAYMENT_METHOD_LABELS[form.method] ?? form.method}${form.month ? ` (${form.month})` : ""}`,
   652→      })
   653→      qc.invalidateQueries({ queryKey: ["staff"] })
   654→      qc.invalidateQueries({ queryKey: ["expenses"] })
   655→      qc.invalidateQueries({ queryKey: ["payments"] })
   656→      qc.invalidateQueries({ queryKey: ["accounts"] })
   657→      onClose()
   658→    } catch (e: any) {
   659→      toast({ title: "Payment failed", description: e.message, variant: "destructive" })
   660→    } finally {
   661→      setSaving(false)
   662→    }
   663→  }
   664→
   665→  return (
   666→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   667→      <DialogContent className="max-w-lg">
   668→        <DialogHeader>
   669→          <DialogTitle>Pay {employee.name}</DialogTitle>
   670→        </DialogHeader>
   671→        <div className="grid gap-3 sm:grid-cols-2">
   672→          <Field label="Payment type" required>
   673→            <SelectInput
   674→              value={form.type}
   675→              onChange={changeType}
   676→              options={SALARY_PAYMENT_TYPES.map((t) => ({ value: t, label: SALARY_PAYMENT_TYPE_LABELS[t] }))}
   677→            />
   678→          </Field>
   679→          <Field
   680→            label="Amount (₹)"
   681→            required
   682→            hint={form.type === "SALARY" && employee.salary > 0 ? `Monthly salary: ${fmtMoney(employee.salary)}` : undefined}
   683→          >
   684→            <NumberInput value={form.amount} onChange={(v) => setForm({ ...form, amount: v })} min={0} />
   685→          </Field>
   686→          <Field label="Method" required>
   687→            <SelectInput
   688→              value={form.method}
   689→              onChange={(v) => setForm({ ...form, method: v })}
   690→              options={PAYMENT_METHODS.map((m) => ({ value: m, label: PAYMENT_METHOD_LABELS[m] }))}
   691→            />
   692→          </Field>
   693→          <Field label="Month" hint="Salary period (YYYY-MM)">
   694→            <TextInput type="month" value={form.month} onChange={(v) => setForm({ ...form, month: v })} />
   695→          </Field>
   696→          <Field label="Date" required>
   697→            <TextInput type="date" value={form.date} onChange={(v) => setForm({ ...form, date: v })} />
   698→          </Field>
   699→          <div className="sm:col-span-2">
   700→            <Field label="Notes">
   701→              <TextArea value={form.notes} onChange={(v) => setForm({ ...form, notes: v })} rows={2} placeholder="Remarks…" />
   702→            </Field>
   703→          </div>
   704→        </div>
   705→        <p className="rounded-md bg-muted/60 p-2.5 text-xs text-muted-foreground">
   706→          Salary &amp; bonus payments also create an expense entry and a money-out payment record. Advances are recoverable and not expensed; deductions reduce take-home.
   707→        </p>
   708→        <DialogFooter>
   709→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   710→          <Button onClick={save} disabled={saving}>
   711→            {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />} Record Payment
   712→          </Button>
   713→        </DialogFooter>
   714→      </DialogContent>
   715→    </Dialog>
   716→  )
   717→}

   720→function AttendanceTab({ employees, loading: staffLoading }: { employees: Employee[]; loading?: boolean }) {
   721→  const qc = useQueryClient()
   722→  const canEdit = canDo("staff", "edit")
   723→  const [view, setView] = useState<"day" | "month">("day")
   724→  const [date, setDate] = useState(ymdIST())
   725→  const [month, setMonth] = useState(ymIST(0))
   726→  const [entries, setEntries] = useState<Record<string, string>>({})
   727→  const [saving, setSaving] = useState(false)
   728→
   729→  const dayQuery = useQuery({
   730→    queryKey: ["staff", "attendance", "day", date],
   731→    queryFn: () => api.get(`staff/attendance${qs({ date })}`),
   732→    enabled: view === "day" && !!date,
   733→  })
   734→
   735→  // Prefill marks from saved records whenever the day data (re)loads
   736→  useEffect(() => {
   737→    const records: AttendanceRecord[] = dayQuery.data?.records ?? []
   738→    setEntries(Object.fromEntries(records.map((r) => [r.employeeId, r.status])))
   739→  }, [dayQuery.data])
   740→
   741→  const monthQuery = useQuery({
   742→    queryKey: ["staff", "attendance", "month", month],
   743→    queryFn: () => api.get(`staff/attendance${qs({ month })}`),
   744→    enabled: view === "month" && /^\d{4}-\d{2}$/.test(month),
   745→  })
   746→
   747→  // Summary chips
   748→  const counts: Record<string, number> = { PRESENT: 0, ABSENT: 0, HALF_DAY: 0, LEAVE: 0 }
   749→  for (const e of employees) {
   750→    const s = entries[e.id]
   751→    if (s && counts[s] !== undefined) counts[s] += 1
   752→  }
   753→  const marked = ATTENDANCE_STATUSES.reduce((s, k) => s + (counts[k] ?? 0), 0)
   754→  const notMarked = Math.max(0, employees.length - marked)
   755→
   756→  function markAllPresent() {
   757→    setEntries(Object.fromEntries(employees.map((e) => [e.id, "PRESENT"])))
   758→  }
   759→
   760→  async function saveAttendance() {
   761→    const payload = employees
   762→      .filter((e) => entries[e.id])
   763→      .map((e) => ({ employeeId: e.id, status: entries[e.id] }))
   764→    if (payload.length === 0) {
   765→      return toast({ title: "Nothing to save", description: "Mark at least one employee first.", variant: "destructive" })
   766→    }
   767→    setSaving(true)
   768→    try {
   769→      const res = await api.post("staff/attendance", { date, entries: payload })
   770→      const saved: number = res?.records?.length ?? payload.length
   771→      toast({
   772→        title: "Attendance saved",
   773→        description: `${saved} ${saved === 1 ? "employee" : "employees"} marked for ${fmtDateIST(date)}.`,
   774→      })
   775→      qc.invalidateQueries({ queryKey: ["staff"] })
   776→    } catch (e: any) {
   777→      toast({ title: "Failed to save attendance", description: e.message, variant: "destructive" })
   778→    } finally {
   779→      setSaving(false)
   780→    }
   781→  }
   782→
   783→  // Month matrix helpers
   784→  const [year, mon] = month.split("-").map(Number)
   785→  const daysInMonth = new Date(year, mon, 0).getDate()
   786→  const dayMap = useMemo(() => {
   787→    const m = new Map<string, string>()
   788→    for (const r of (monthQuery.data?.records ?? []) as AttendanceRecord[]) {
   789→      m.set(`${r.employeeId}|${parseInt(r.date.slice(8), 10)}`, r.status)
   790→    }
   791→    return m
   792→  }, [monthQuery.data])
   793→
   794→  function isSunday(day: number) {
   795→    return new Date(Date.UTC(year, mon - 1, day)).getUTCDay() === 0
   796→  }
   797→
   798→  return (
   799→    <div className="space-y-4">
   800→      {/* Toolbar */}
   801→      <div className="flex flex-wrap items-center gap-2">
   802→        {view === "day" ? (
   803→          <div className="relative">
   804→            <CalendarCheck className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   805→            <Input
   806→              type="date"
   807→              aria-label="Attendance date"
   808→              value={date}
   809→              onChange={(e) => setDate(e.target.value)}
   810→              className="h-9 w-[170px] pl-9"
   811→            />
   812→          </div>
   813→        ) : (
   814→          <div className="relative">
   815→            <CalendarDays className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   816→            <Input
   817→              type="month"
   818→              aria-label="Attendance month"
   819→              value={month}
   820→              onChange={(e) => setMonth(e.target.value)}
   821→              className="h-9 w-[170px] pl-9"
   822→            />
   823→          </div>
   824→        )}
   825→        {view === "day" && canEdit && (
   826→          <>
   827→            <Button variant="outline" size="sm" className="h-9" onClick={markAllPresent} disabled={employees.length === 0}>
   828→              <CheckCheck className="mr-1.5 h-4 w-4" /> All Present

   830→            <Button size="sm" className="h-9" onClick={saveAttendance} disabled={saving || !date || employees.length === 0}>

   832→              Save Attendance
