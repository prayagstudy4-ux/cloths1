    84→    84→/** Current IST month as YYYY-MM, offset by n months */
    85→    85→function ymIST(offsetMonths = 0): string {
    86→    86→  const [y, m] = ymdIST().split("-").map(Number)
    87→    87→  let year = y, month = m + offsetMonths
    88→    88→  while (month < 1) { month += 12; year -= 1 }
    89→    89→  while (month > 12) { month -= 12; year += 1 }
    90→    90→  return `${year}-${String(month).padStart(2, "0")}`
    91→    91→}

    93→    93→function isTaskOverdue(t: Task): boolean {
    94→    94→  if (!t.dueDate || t.status === "COMPLETED" || t.status === "CANCELLED") return false
    95→    95→  return fmtDateIST(t.dueDate, "yyyy-MM-dd") < ymdIST()
    96→    96→}

   280→   280→function EmployeeForm({ employee, onClose }: { employee?: Employee; onClose: () => void }) {
   281→   281→  const qc = useQueryClient()
   282→   282→  const [saving, setSaving] = useState(false)
   283→   283→  const [form, setForm] = useState({
   284→   284→    name: employee?.name ?? "",
   285→   285→    phone: employee?.phone ?? "",
   286→   286→    designation: employee?.designation ?? "",
   287→   287→    joiningDate: employee?.joiningDate ? fmtDateIST(employee.joiningDate, "yyyy-MM-dd") : ymdIST(),
   288→   288→    salary: employee?.salary ?? 0,
   289→   289→    address: employee?.address ?? "",
   290→   290→    notes: employee?.notes ?? "",
   291→   291→  })
   292→   292→
   293→   293→  async function save() {
   294→   294→    if (!form.name.trim()) return toast({ title: "Name is required", variant: "destructive" })
   295→   295→    setSaving(true)
   296→   296→    try {
   297→   297→      const payload = {
   298→   298→        name: form.name.trim(),
   299→   299→        phone: form.phone.trim(),
   300→   300→        designation: form.designation.trim(),
   301→   301→        joiningDate: form.joiningDate || undefined,
   302→   302→        salary: form.salary,
   303→   303→        address: form.address.trim(),
   304→   304→        notes: form.notes.trim(),
   305→   305→      }
   306→   306→      if (employee) await api.put(`staff/${employee.id}`, payload)
   307→   307→      else await api.post("staff", payload)
   308→   308→      toast({
   309→   309→        title: employee ? "Employee updated" : "Employee created",
   310→   310→        description: `${form.name.trim()}${form.designation.trim() ? ` · ${form.designation.trim()}` : ""}`,
   311→   311→      })
   312→   312→      qc.invalidateQueries({ queryKey: ["staff"] })
   313→   313→      onClose()
   314→   314→    } catch (e: any) {
   315→   315→      toast({ title: "Failed to save employee", description: e.message, variant: "destructive" })
   316→   316→    } finally {
   317→   317→      setSaving(false)
   318→   318→    }
   319→   319→  }
   320→   320→
   321→   321→  return (
   322→   322→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   323→   323→      <DialogContent className="max-w-lg">
   324→   324→        <DialogHeader>
   325→   325→          <DialogTitle>{employee ? `Edit ${employee.name}` : "New Employee"}</DialogTitle>
   326→   326→        </DialogHeader>
   327→   327→        <div className="grid gap-3 sm:grid-cols-2">
   328→   328→          <div className="sm:col-span-2">
   329→   329→            <Field label="Full name" required>
   330→   330→              <TextInput value={form.name} onChange={(v) => setForm({ ...form, name: v })} placeholder="Employee name" autoFocus />
   331→   331→            </Field>
   332→   332→          </div>
   333→   333→          <Field label="Phone">
   334→   334→            <TextInput value={form.phone} onChange={(v) => setForm({ ...form, phone: v })} placeholder="98XXXXXXXX" />
   335→   335→          </Field>
   336→   336→          <Field label="Designation" hint="e.g. Sales Person, Tailor, Helper">
   337→   337→            <TextInput value={form.designation} onChange={(v) => setForm({ ...form, designation: v })} placeholder="Designation" />
   338→   338→          </Field>
   339→   339→          <Field label="Joining date" hint={employee ? "Server keeps the original joining date for now" : undefined}>
   340→   340→            <TextInput type="date" value={form.joiningDate} onChange={(v) => setForm({ ...form, joiningDate: v })} />
   341→   341→          </Field>
   342→   342→          <Field label="Monthly salary (₹)">
   343→   343→            <NumberInput value={form.salary} onChange={(v) => setForm({ ...form, salary: v })} min={0} />
   344→   344→          </Field>
   345→   345→          <div className="sm:col-span-2">
   346→   346→            <Field label="Address">
   347→   347→              <TextInput value={form.address} onChange={(v) => setForm({ ...form, address: v })} placeholder="Street, city…" />
   348→   348→            </Field>
   349→   349→          </div>
   350→   350→          <div className="sm:col-span-2">
   351→   351→            <Field label="Notes">
   352→   352→              <TextArea value={form.notes} onChange={(v) => setForm({ ...form, notes: v })} rows={2} placeholder="Remarks, skills, references…" />
   353→   353→            </Field>
   354→   354→          </div>
   355→   355→        </div>
   356→   356→        <DialogFooter>
   357→   357→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   358→   358→          <Button onClick={save} disabled={saving}>
   359→   359→            {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}Save
   360→   360→          </Button>
   361→   361→        </DialogFooter>
   362→   362→      </DialogContent>
   363→   363→    </Dialog>
   364→   364→  )

   381→   381→  const [togglingStatus, setTogglingStatus] = useState(false)
   382→   382→
   383→   383→  // Salary / advance history (documented endpoint: GET staff/payments?employeeId=)
   384→   384→  const { data: payData, isLoading: payLoading } = useQuery({
   385→   385→    queryKey: ["staff", "payments", employeeId],
   386→   386→    queryFn: () => api.get(`staff/payments${qs({ employeeId })}`),
   387→   387→  })
   388→   388→
   389→   389→  // Recent attendance = current + previous month via GET staff/attendance?month=
   390→   390→  const curMonth = useMemo(() => ymIST(0), [])

   626→   626→function PaySalaryDialog({ employee, onClose }: { employee: Employee; onClose: () => void }) {

   720→   720→function AttendanceTab({ employees, loading: staffLoading }: { employees: Employee[]; loading?: boolean }) {
   721→   721→  const qc = useQueryClient()
   722→   722→  const canEdit = canDo("staff", "edit")
   723→   723→  const [view, setView] = useState<"day" | "month">("day")
   724→   724→  const [date, setDate] = useState(ymdIST())
   725→   725→  const [month, setMonth] = useState(ymIST(0))
   726→   726→  const [entries, setEntries] = useState<Record<string, string>>({})
   727→   727→  const [saving, setSaving] = useState(false)
   728→   728→
   729→   729→  const dayQuery = useQuery({
   730→   730→    queryKey: ["staff", "attendance", "day", date],
   731→   731→    queryFn: () => api.get(`staff/attendance${qs({ date })}`),
   732→   732→    enabled: view === "day" && !!date,
   733→   733→  })
   734→   734→
   735→   735→  // Prefill marks from saved records whenever the day data (re)loads
   736→   736→  useEffect(() => {
   737→   737→    const records: AttendanceRecord[] = dayQuery.data?.records ?? []
   738→   738→    setEntries(Object.fromEntries(records.map((r) => [r.employeeId, r.status])))
   739→   739→  }, [dayQuery.data])
   740→   740→
   741→   741→  const monthQuery = useQuery({
   742→   742→    queryKey: ["staff", "attendance", "month", month],
   743→   743→    queryFn: () => api.get(`staff/attendance${qs({ month })}`),
   744→   744→    enabled: view === "month" && /^\d{4}-\d{2}$/.test(month),
   745→   745→  })
   746→   746→
   747→   747→  // Summary chips
   748→   748→  const counts: Record<string, number> = { PRESENT: 0, ABSENT: 0, HALF_DAY: 0, LEAVE: 0 }
   749→   749→  for (const e of employees) {
   750→   750→    const s = entries[e.id]
   751→   751→    if (s && counts[s] !== undefined) counts[s] += 1
   752→   752→  }
   753→   753→  const marked = ATTENDANCE_STATUSES.reduce((s, k) => s + (counts[k] ?? 0), 0)
   754→   754→  const notMarked = Math.max(0, employees.length - marked)
   755→   755→
   756→   756→  function markAllPresent() {
   757→   757→    setEntries(Object.fromEntries(employees.map((e) => [e.id, "PRESENT"])))
   758→   758→  }
   759→   759→
   760→   760→  async function saveAttendance() {
   761→   761→    const payload = employees
   762→   762→      .filter((e) => entries[e.id])
   763→   763→      .map((e) => ({ employeeId: e.id, status: entries[e.id] }))
   764→   764→    if (payload.length === 0) {
   765→   765→      return toast({ title: "Nothing to save", description: "Mark at least one employee first.", variant: "destructive" })
   766→   766→    }
   767→   767→    setSaving(true)
   768→   768→    try {
   769→   769→      const res = await api.post("staff/attendance", { date, entries: payload })
   770→   770→      const saved: number = res?.records?.length ?? payload.length
