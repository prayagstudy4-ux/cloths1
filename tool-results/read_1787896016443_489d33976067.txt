   870→// ==================== NEW PRODUCTION ORDER DIALOG ====================
   871→function NewProductionOrderDialog({ onClose }: { onClose: () => void }) {
   872→  const qc = useQueryClient()
   873→  const { data: productsData, isLoading: productsLoading } = useQuery({
   874→    queryKey: ["products", "lookup"],
   875→    queryFn: () => api.get("products?pageSize=200"),
   876→  })
   877→  const { data: contractorsData } = useQuery({ queryKey: ["contractors"], queryFn: () => api.get("contractors") })
   878→
   879→  const products: ProductLite[] = productsData?.products ?? []
   880→  const contractors: Contractor[] = contractorsData?.contractors ?? []
   881→
   882→  const [form, setForm] = useState({
   883→    productId: "", designName: "", quantity: 100, contractorId: "", targetDate: "", costEstimate: 0, notes: "",
   884→  })
   885→  const [planQty, setPlanQty] = useState<Record<string, number>>({})
   886→  const [saving, setSaving] = useState(false)
   887→
   888→  const product = products.find((p) => p.id === form.productId)
   889→  const variants = product?.variants ?? []
   890→  const plannedTotal = Object.values(planQty).reduce((s, n) => s + n, 0)
   891→  const multiVariant = variants.length > 1
   892→
   893→  async function save() {
   894→    if (!form.productId) return toast({ title: "Product is required", variant: "destructive" })
   895→    if (!form.quantity || form.quantity < 1) return toast({ title: "Quantity must be at least 1", variant: "destructive" })
   896→    setSaving(true)
   897→    try {
   898→      const body: Record<string, unknown> = {
   899→        productId: form.productId,
   900→        quantity: Math.round(form.quantity),
   901→        designName: form.designName || undefined,
   902→        contractorId: form.contractorId || undefined,
   903→        targetDate: form.targetDate || undefined,
   904→        costEstimate: form.costEstimate || undefined,
   905→        notes: form.notes || undefined,
   906→      }
   907→      if (variants.length === 1) {
   908→        body.planLines = [{ variantId: variants[0].id, qty: Math.round(form.quantity) }]
   909→      } else if (multiVariant) {
   910→        const lines = Object.entries(planQty).filter(([, qty]) => qty > 0).map(([variantId, qty]) => ({ variantId, qty: Math.round(qty) }))
   911→        if (lines.length) body.planLines = lines
   912→      }
   913→      const res = await api.post("production", body)
   914→      toast({ title: `Production order ${res?.order?.number ?? ""} created`, description: "Track its progress from the Production Orders tab." })
   915→      qc.invalidateQueries({ queryKey: ["production"] })
   916→      onClose()
   917→    } catch (e: any) {
   918→      toast({ title: "Failed to create order", description: e.message, variant: "destructive" })
   919→    } finally {
   920→      setSaving(false)
   921→    }
   922→  }
   923→
   924→  return (
   925→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   926→      <DialogContent className="max-w-lg">
   927→        <DialogHeader>
   928→          <DialogTitle>New Production Order</DialogTitle>
   929→        </DialogHeader>
   930→        <div className="grid gap-3 sm:grid-cols-2">
   931→          <div className="sm:col-span-2">
   932→            <Field label="Product" required hint={productsLoading ? "Loading products…" : undefined}>
   933→              <EntityPicker
