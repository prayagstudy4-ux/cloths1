   114→function parsePlan(o: ProductionOrder): PlanLine[] {
   115→  const raw = (o as any).plan ?? o.planLines
   116→  if (Array.isArray(raw)) return raw as PlanLine[]
   117→  if (typeof raw === "string" && raw) {
   118→    try { return JSON.parse(raw) as PlanLine[] } catch { return [] }
   119→  }
   120→  return []
   121→}

   123→function variantShortLabel(v: VariantLite): string {
   124→  const parts = [v.size?.name, v.color?.name].filter(Boolean).join(" · ")

   141→// Compact horizontal stage indicator for order cards
   142→function StageDots({ stage, status }: { stage: string; status: string }) {
   143→  const idx = STAGES.indexOf(stage)
   144→  const done = status === "COMPLETED"
   145→  const cancelled = status === "CANCELLED"
   146→  const pct = done ? 100 : Math.round((Math.max(0, idx) / (STAGES.length - 1)) * 100)
   147→  return (
   148→    <div>
   149→      <div className="flex items-center" aria-label={`Stage ${idx + 1} of ${STAGES.length}: ${PRODUCTION_STAGE_LABELS[stage] ?? stage}`}>
   150→        {STAGES.map((s, i) => (
   151→          <Fragment key={s}>
   152→            {i > 0 && <div className={cn("h-0.5 min-w-[6px] flex-1", !cancelled && (done || i <= idx) ? "bg-primary" : "bg-muted")} />}
   153→            <div
   154→              className={cn(
   155→                "h-2 w-2 shrink-0 rounded-full transition-colors",
   156→                cancelled ? "bg-muted-foreground/40" :
   157→                  done || i < idx ? "bg-primary" :
   158→                    i === idx ? "bg-primary ring-2 ring-primary/25 ring-offset-1" : "bg-muted-foreground/30",
   159→              )}
   160→            />
   161→          </Fragment>
   162→        ))}
   163→      </div>
   164→      <div className="mt-1.5 flex items-center justify-between text-[11px]">
   165→        <span className="font-medium">
   166→          {cancelled ? `Cancelled at ${PRODUCTION_STAGE_LABELS[stage] ?? stage}` :
   167→            done ? "All stages complete" :
   168→              `Stage ${idx + 1}/${STAGES.length} · ${PRODUCTION_STAGE_LABELS[stage] ?? stage}`}
   169→        </span>
   170→        <span className="tabular-nums text-muted-foreground">{pct}%</span>
   171→      </div>
   172→    </div>
   173→  )
   174→}

   176→// Full vertical stepper for the order detail sheet
   177→function StageStepper({ stage, status }: { stage: string; status: string }) {
   178→  const idx = STAGES.indexOf(stage)
   179→  return (
   180→    <ol className="relative">
   181→      {STAGES.map((s, i) => {
   182→        const isDone = status === "COMPLETED" || i < idx
   183→        const isCurrent = status === "IN_PROGRESS" && i === idx
   184→        const isCancelledAt = status === "CANCELLED" && i === idx
   185→        return (
   186→          <li key={s} className="relative flex gap-3 pb-3.5 last:pb-0">
   187→            {i < STAGES.length - 1 && (
   188→              <span className={cn("absolute left-[11px] top-7 h-[calc(100%-16px)] w-0.5", isDone ? "bg-primary/50" : "bg-border")} />
   189→            )}
   190→            <span
   191→              className={cn(
   192→                "relative z-10 flex h-6 w-6 shrink-0 items-center justify-center rounded-full text-[11px] font-semibold",
   193→                isCancelledAt ? "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300" :
   194→                  isDone ? "bg-primary text-primary-foreground" :
   195→                    isCurrent ? "border-2 border-primary bg-primary/10 text-primary" :
   196→                      "bg-muted text-muted-foreground",
   197→              )}
   198→            >
   199→              {isDone ? <Check className="h-3.5 w-3.5" /> : isCancelledAt ? <X className="h-3.5 w-3.5" /> : i + 1}
   200→            </span>
   201→            <div className="min-w-0 pt-0.5">
   202→              <p className={cn("text-sm leading-6", isCurrent || isCancelledAt ? "font-semibold" : "text-muted-foreground")}>
   203→                {PRODUCTION_STAGE_LABELS[s] ?? s}
   204→              </p>
   205→              {isCurrent && <p className="text-xs font-medium text-primary">Current stage</p>}
   206→              {isCancelledAt && <p className="text-xs text-red-600 dark:text-red-400">Order cancelled at this stage</p>}
   207→            </div>
   208→          </li>
   209→        )
   210→      })}
   211→    </ol>
   212→  )
   213→}

   215→// ==================== MODULE ROOT ====================
   216→export function ProductionModule() {
   217→  const { moduleParams } = useApp()
   218→  const qc = useQueryClient()
   219→  const [tab, setTab] = useState(() => normalizeTab(moduleParams?.tab))
   220→
   221→  // detail sheet + deep-link resolution
   222→  const [detail, setDetail] = useState<DetailRef | null>(null)
   223→  const [resolveId, setResolveId] = useState<string | null>(null)
   224→  const [materialEditId, setMaterialEditId] = useState<string | null>(null)
   225→
   226→  // dialog states (root-level so any tab / detail can trigger them)
   227→  const [newOrder, setNewOrder] = useState(false)
   228→  const [newJobWork, setNewJobWork] = useState<{ contractorId?: string } | null>(null)
   229→  const [newContractor, setNewContractor] = useState(false)
   230→  const [newMaterial, setNewMaterial] = useState(false)
   231→  const [editingContractor, setEditingContractor] = useState<Contractor | null>(null)
   232→  const [editingMaterial, setEditingMaterial] = useState<RawMaterial | null>(null)
   233→  const [addStockTo, setAddStockTo] = useState<RawMaterial | null>(null)
   234→  const [deletingContractor, setDeletingContractor] = useState<Contractor | null>(null)
   235→  const [deletingMaterial, setDeletingMaterial] = useState<RawMaterial | null>(null)
   236→
   237→  // Apply module params (deep links from sidebar, command palette, other modules) once per
   238→  // distinct params object. Deferred a tick so we never cascade renders synchronously.
   239→  const appliedRef = useRef<string | null>(null)
   240→  useEffect(() => {
   241→    if (!moduleParams) return
   242→    const key = JSON.stringify(moduleParams)
   243→    if (appliedRef.current === key) return
   244→    appliedRef.current = key
   245→    const timer = setTimeout(() => {
   246→      const t = moduleParams.tab ? normalizeTab(moduleParams.tab) : null
   247→      if (t) setTab(t)
   248→      const eid = moduleParams.entityId as string | undefined
   249→      if (eid) {
   250→        if (t === "orders") setDetail({ type: "order", id: eid })
   251→        else if (t === "jobwork") setDetail({ type: "jobwork", id: eid })
   252→        else if (t === "contractors") setDetail({ type: "contractor", id: eid })
   253→        else if (t === "materials") setMaterialEditId(eid)
   254→        else setResolveId(eid)
   255→      }
   256→      if (moduleParams.new) {
   257→        if (t === "jobwork") setNewJobWork({})
   258→        else if (t === "contractors") setNewContractor(true)
   259→        else if (t === "materials") setNewMaterial(true)
   260→        else setNewOrder(true)
   261→      }
   262→    }, 0)
   263→    return () => clearTimeout(timer)
   264→  }, [moduleParams])
   265→
   266→  // Data queries — all lists are fetched unfiltered so summary stats stay stable
   267→  const ordersQuery = useQuery({
   268→    queryKey: ["production"],
   269→    queryFn: () => api.get("production"),
   270→    enabled: tab === "orders",
   271→  })
   272→  const jobworksQuery = useQuery({
   273→    queryKey: ["jobworks"],
   274→    queryFn: () => api.get("jobworks"),
   275→    enabled: tab === "jobwork",
   276→  })
   277→  const contractorsQuery = useQuery({
   278→    queryKey: ["contractors"],
   279→    queryFn: () => api.get("contractors"),
   280→    enabled: tab === "contractors",
   281→  })
   282→  const materialsQuery = useQuery({
   283→    queryKey: ["raw-materials"],
   284→    queryFn: () => api.get("raw-materials"),
   285→    enabled: tab === "materials",
   286→  })
   287→  const productsQuery = useQuery({
   288→    queryKey: ["products", "lookup"],
   289→    queryFn: () => api.get("products?pageSize=200"),
   290→    enabled: tab === "orders" || tab === "jobwork" || !!detail,
   291→  })
   292→
   293→  // Product / variant lookups (orders & job works only expose productId)
   294→  const products: ProductLite[] = useMemo(() => productsQuery.data?.products ?? [], [productsQuery.data])
   295→  const productById = useMemo(() => new Map(products.map((p) => [p.id, p])), [products])
   296→  const variantById = useMemo(() => {
   297→    const m = new Map<string, { variant: VariantLite; product: ProductLite }>()
   298→    for (const p of products) for (const v of p.variants ?? []) m.set(v.id, { variant: v, product: p })
   299→    return m
   300→  }, [products])
   301→  const productName = useCallback(
   302→    (id: string | null | undefined) => (id ? productById.get(id)?.name ?? "—" : "—"),
   303→    [productById],
   304→  )
   305→  const variantLabel = useCallback(
   306→    (id: string | null | undefined) => {
   307→      if (!id) return "—"
   308→      const e = variantById.get(id)
   309→      if (!e) return "Linked variant"
   310→      return `${e.product.name} — ${variantShortLabel(e.variant)}`
   311→    },
   312→    [variantById],
   313→  )
   314→
   315→  // Resolve an entityId that arrived without a tab (e.g. Ctrl+K global search hits
   316→  // contractors / job works under the production module). Loads the shared list queries
   317→  // through the cache so the resolved tab renders instantly afterwards.
   318→  useEffect(() => {
   319→    if (!resolveId) return
   320→    let cancelled = false
   321→    ;(async () => {
   322→      try {
   323→        const [prodRes, jwRes, conRes, matRes] = await Promise.all([
   324→          qc.ensureQueryData({ queryKey: ["production"], queryFn: () => api.get("production") }),
   325→          qc.ensureQueryData({ queryKey: ["jobworks"], queryFn: () => api.get("jobworks") }),
   326→          qc.ensureQueryData({ queryKey: ["contractors"], queryFn: () => api.get("contractors") }),
   327→          qc.ensureQueryData({ queryKey: ["raw-materials"], queryFn: () => api.get("raw-materials") }),
   328→        ] as const)
   329→        if (cancelled) return
   330→        const orders: any[] = (prodRes as any)?.orders ?? []
   331→        const jws: any[] = (jwRes as any)?.jobWorks ?? []
   332→        const cons: any[] = (conRes as any)?.contractors ?? []
   333→        const mats: any[] = (matRes as any)?.materials ?? []
   334→        if (orders.some((o) => o.id === resolveId)) { setDetail({ type: "order", id: resolveId }); setTab("orders") }
   335→        else if (jws.some((j) => j.id === resolveId)) { setDetail({ type: "jobwork", id: resolveId }); setTab("jobwork") }
   336→        else if (cons.some((c) => c.id === resolveId)) { setDetail({ type: "contractor", id: resolveId }); setTab("contractors") }
   337→        else if (mats.some((m) => m.id === resolveId)) { setMaterialEditId(resolveId); setTab("materials") }
   338→        else {
   339→          toast({ title: "Record not found", description: "The linked production record no longer exists.", variant: "destructive" })
   340→        }
   341→      } catch {
   342→        // network/permission hiccup — the module stays usable
   343→      } finally {
   344→        if (!cancelled) setResolveId(null)
   345→      }
   346→    })()
   347→    return () => { cancelled = true }
   348→  }, [resolveId, qc])
   349→
   350→  const canCreate = canDo("production", "create")
   351→  const canEdit = canDo("production", "edit")
   352→
   353→  const primaryAction =
   354→    tab === "jobwork" ? { label: "New Job Work", icon: <Scissors className="mr-1.5 h-4 w-4" />, onClick: () => setNewJobWork({}) } :
   355→      tab === "contractors" ? { label: "New Contractor", icon: <HardHat className="mr-1.5 h-4 w-4" />, onClick: () => setNewContractor(true) } :
   356→        tab === "materials" ? { label: "New Material", icon: <PackagePlus className="mr-1.5 h-4 w-4" />, onClick: () => setNewMaterial(true) } :
   357→          { label: "New Production Order", icon: <Plus className="mr-1.5 h-4 w-4" />, onClick: () => setNewOrder(true) }
   358→
   359→  return (
   360→    <div className="space-y-4">
   361→      <PageHeader
   362→        icon={<Factory className="h-5 w-5" />}
   363→        title="Production"
   364→        description="Track production orders from design to packaging, manage contractor job work and raw material stock."
   365→        actions={
   366→          <>
   367→            <Button
   368→              variant="outline" size="sm"
   369→              onClick={() => {
   370→                qc.invalidateQueries({ queryKey: ["production"] })
   371→                qc.invalidateQueries({ queryKey: ["jobworks"] })
   372→                qc.invalidateQueries({ queryKey: ["contractors"] })
   373→                qc.invalidateQueries({ queryKey: ["raw-materials"] })
   374→              }}
   375→            >
   376→              Refresh
   377→            </Button>
   378→            {canCreate && (
   379→              <Button size="sm" onClick={primaryAction.onClick}>
   380→                {primaryAction.icon} {primaryAction.label}
   381→              </Button>
   382→            )}
   383→          </>
   384→        }
   385→      />
   386→
   387→      <Tabs value={tab} onValueChange={setTab}>
   388→        <TabsList className="w-full justify-start overflow-x-auto flex-wrap h-auto">
   389→          <TabsTrigger value="orders" className="gap-1.5"><Factory className="h-4 w-4" /> Production Orders</TabsTrigger>
   390→          <TabsTrigger value="jobwork" className="gap-1.5"><Scissors className="h-4 w-4" /> Job Work</TabsTrigger>
   391→          <TabsTrigger value="contractors" className="gap-1.5"><HardHat className="h-4 w-4" /> Contractors</TabsTrigger>
   392→          <TabsTrigger value="materials" className="gap-1.5"><Package className="h-4 w-4" /> Raw Materials</TabsTrigger>
   393→        </TabsList>
   394→
   395→        <TabsContent value="orders" className="mt-4">
   396→          <OrdersTab
   397→            orders={ordersQuery.data?.orders ?? []}
   398→            summary={ordersQuery.data?.summary}
   399→            loading={ordersQuery.isLoading}
   400→            productName={productName}
   401→            onOpen={(id) => setDetail({ type: "order", id })}
   402→            onNew={() => setNewOrder(true)}
   403→            canCreate={canCreate}
   404→          />
   405→        </TabsContent>
   406→
   407→        <TabsContent value="jobwork" className="mt-4">
   408→          <JobWorkTab
   409→            jobWorks={jobworksQuery.data?.jobWorks ?? []}
   410→            summary={jobworksQuery.data?.summary}
   411→            loading={jobworksQuery.isLoading}
   412→            variantLabel={variantLabel}
   413→            onOpen={(id) => setDetail({ type: "jobwork", id })}
   414→            onNew={() => setNewJobWork({})}
   415→            canCreate={canCreate}
   416→          />
   417→        </TabsContent>
   418→
   419→        <TabsContent value="contractors" className="mt-4">
   420→          <ContractorsTab
   421→            contractors={contractorsQuery.data?.contractors ?? []}
   422→            loading={contractorsQuery.isLoading}
   423→            onOpen={(id) => setDetail({ type: "contractor", id })}
   424→            onNew={() => setNewContractor(true)}
   425→            canCreate={canCreate}
   426→          />
   427→        </TabsContent>
   428→
   429→        <TabsContent value="materials" className="mt-4">
